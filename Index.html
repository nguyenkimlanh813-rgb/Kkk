using UnityEngine; public class TrafficVehicle : MonoBehaviour { public float speed = 10f; // Tốc độ của xe giao thông public bool moveOpposite = true; // Đúng = chạy ngược chiều với người chơi void Update() { // Nếu moveOpposite = true, xe chạy ngược lại (hướng Vector3.back) Vector3 direction = moveOpposite ? Vector3.back : Vector3.forward; transform.Translate(direction * speed * Time.deltaTime); // Tự động hủy xe nếu chạy quá xa để tránh nặng máy (sau khi đi qua người chơi) if (transform.position.z < -20f) { Destroy(gameObject); } } }
